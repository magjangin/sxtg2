**개선안**:
```csharp
private void HandleSceneChange(string sceneName, int sceneIndex, bool isInitialLoad = false)
{
    if (!string.IsNullOrEmpty(_currentSceneName) && !isInitialLoad)
    {
        MelonLogger.Msg($"[씬 변경] {_currentSceneName} → {sceneName} (인덱스: {sceneIndex})");
    }
    else
    {
        MelonLogger.Msg($"[씬 감지] 현재 씬: {sceneName} (인덱스: {sceneIndex})");
    }

    _currentSceneName = sceneName;
    sxtg2.Features.MusicSelectAnalyzer.ResetAnalysisFlag();
    
    if (IsMusicSelectScene(sceneName))
    {
        MelonLogger.Msg("[디버그] MusicSelect 씬 감지, 기준 지연 설정");
        _musicSelectAnalysisDelay = DEFAULT_SCENE_DELAY;
    }
}

private bool IsMusicSelectScene(string sceneName)
{
    return sceneName.Contains("MusicSelect") || sceneName == "MusicSelect";
}
```
